; Имена, найденные при разборе. ЭТОТ файл правите вы, он под git.
;
; game_symbols.gen.txt перезаписывается анализатором при каждой смене ROM —
; туда писать нельзя. Сборочный game_symbols.txt склеивается из двух этих
; файлов (tools/symbols.py --merge), пользовательский идёт первым и
; побеждает: в types/symbol.go выигрывает первое вхождение адреса.
;
; Формат:  Имя = $адрес        (принимаются и `$00A000 Имя`, и `00A000:Имя`)
;
; Можно называть ЛЮБОЙ адрес, не только код: переменные в ОЗУ, таблицы
; внутри bin-сегментов, регистры. Для тех, где метку никто не печатает,
; tools/symbols.py --equates автоматически допишет `equ` в ports.asm.
; ═══════════════════════════════════════════════════════════════════════

; ── Собственный BIOS игры (вызовы через line-F трапы) ──────────────────
LineFDispatch   = $000668   ; читает опкод $FFxx через PC в стеке
LineFJumpTable  = $000690   ; 57 записей, смещения относительно себя

; ── ОЗУ ────────────────────────────────────────────────────────────────
; Состояние геймпадов: пишется единственным местом чтения ввода во всей
; игре, внутри VBlank ($000488). Два байта на порт.
InputState      = $FF003E

; ═══════════════════════════════════════════════════════════════════════
; Найдено разбором кода. Каждое имя — с обоснованием; где вывод неполный,
; это сказано прямо.
; ═══════════════════════════════════════════════════════════════════════

; ── Блок таймеров, обновляется в VBlank ($000430) ─────────────────────
; Обработчик берёт базу `lea ($FF005A).l,a0` и идёт по блоку через (a0)+,
; поэтому все четыре лежат подряд именно в этом порядке.
FrameCounter     = $FF005A   ; word, +1 каждый кадр безусловно
FrameWait        = $FF005C   ; word, +1 каждый кадр. Идиома по всему коду:
                             ; clr.w, затем ждать cmpi.w #$0001 — «ждать кадр»
ElapsedFrames    = $FF005E   ; long, +1 только если бит 7 GlobalFlags сброшен.
                             ; Сравнивается с #$BB8 (3000 кадров ≈ 50 с)
DelayTimer       = $FF0062   ; word с насыщением: при переполнении ставится
                             ; $FFFF. Сброс и ожидание 25/35/40 кадров

; Счёт 0..3: addq.b #1, затем bclr #2 — бит 2 работает как перенос, и на
; каждом четвёртом кадре VBlank выходит раньше. Других обращений в коде нет.
VBlankPhase      = $FF0044

; ── Глобальное состояние ──────────────────────────────────────────────
; Байт битовых флагов, 68 обращений: биты 3,4,6,7 ставятся и проверяются
; поодиночке, clr.b сбрасывает всё. Бит 7 останавливает счёт таймеров.
GlobalFlags      = $FF027E

; Указатель-константа: пишется ровно один раз (move.l #$00FF028A на $006C0A),
; читается 42 раза через movea.l. Значит $FF028A — постоянная база структуры.
MainStatePtr     = $FF0270
MainState        = $FF028A

; Тик игрового времени. Инкремент в $01630A, сброс в $016568, ~90 чтений.
; Сравнивается с конкретными значениями (например #$189C = 6300) — по
; совпадению запускается событие. Это расписание, а не ГПСЧ.
GameTick         = $FFE05C

; Курсор по массиву записей: инициализируется значением $00FF9C84
; ($0164B0), дальше переприсваивается из a0 — то есть двигается.
CurrentRecordPtr = $FF9C7E
RecordArrayBase  = $FF9C84

; Рабочий буфер в ОЗУ, 401 обращение. Через него же идёт распаковка:
; адрес кладут на стек перед вызовом $FF10, а затем $FF04 гонит оттуда в VRAM.
WorkBuffer       = $FF0564

; ── ГПСЧ ──────────────────────────────────────────────────────────────
; Random: 146 вызовов, второй по частоте в игре. Полностью детерминирован:
;   d0 = RandomTable[RandomIndex++] ^ RandomRound
;   RandomRound++ при переполнении RandomIndex (каждые 256 обращений)
; RandomTable — перестановка: все 256 значений байта ровно по разу,
; сумма 32640. Значит последовательность оригинала воспроизводима точно.
Random           = $016570
RandomTable      = $0165B2
RandomIndex      = $FF054E
RandomRound      = $FF054F

; ── Инициализация ($0008F8) ───────────────────────────────────────────
; Теневая копия регистров VDP: цикл на 19 итераций читает значения из ROM,
; кладёт их сюда и одновременно пишет в VDP_CTRL. Занимает $FF0018-$FF003D
; (19 слов) и кончается ровно там, где начинается InputState.
VDPRegShadow     = $FF0018

; Ставится один раз при старте (move.b #$60) и больше не меняется; дальше
; только btst битов 5, 6, 7 в 17 местах. Похоже на конфигурацию железа,
; но что именно кодируют биты — из кода не видно.
ConfigFlags      = $FF0283

; ── Точки входа BIOS игры (вызываются через line-F трапы) ─────────────
; Аргументы кладутся на стек вызывающим; обработчик читает их по a6,
; который диспетчер настраивает на кадр вызова.

; $FF10 — Decompress(dest, src). Сам диспетчер: байт +2 заголовка блока
; выбирает метод из таблицы $000D5A (реально 7 методов, в данных почти
; всегда метод 7 = $000F3A).
BiosDecompress   = $000D42

; $FF04 — CopyToVRAM(vramAddr, src, len). Собирает командный лонг VDP:
; (addr & $3FFF) << 16 | (addr >> 14) & 3 | $40000000.
BiosCopyToVRAM   = $000AF8

; $FF03 — FillVRAM(cmd, len, value).
BiosFillVRAM     = $000C18

; $FF20 — LoadPalette(line, src): запись в CRAM со смещением line*32,
; то есть по 16 цветов на строку палитры.
BiosLoadPalette  = $001316

; $FF21 / $FF22 — очистка таблиц имён: обе зовут FillVRAM нулями на $1000
; байт, но с адресами $C000 и $E000 — это плоскости A и B.
BiosClearPlaneA  = $00134C
BiosClearPlaneB  = $001356

; Самая вызываемая процедура игры — 393 вызова. Читает статус VDP и
; проверяет бит 3 (флаг VBlank). Ожидание кадра.
VDPWaitVBlank    = $000AB6

; ── Звук ──────────────────────────────────────────────────────────────
; $FF2F — PlaySound(id). Проверяет бит 4 GlobalFlags (глушение), отбрасывает
; id >= $E0 и кладёт байт в SoundRequest. Цепочка доказана до конца: VBlank
; ($0005AE) читает SoundRequest и пишет его в ОЗУ Z80 по $A01C0A с полным
; рукопожатием по шине, затем очищает. Значит бит 4 GlobalFlags = звук выкл.
BiosPlaySound    = $001E04
SoundRequest     = $FF0282
Z80SoundCommand  = $A01C0A   ; почтовый ящик в ОЗУ Z80: команда драйверу
Z80SoundStatus   = $A01C08   ; ответ драйвера, читается в VBlank

; ── Ещё точки входа BIOS ──────────────────────────────────────────────
; $FF01 — тот же копировщик, что $FF04, но принимает ГОТОВЫЙ командный лонг
; VDP: $0(a6) команда, $4(a6) источник, $8(a6) длина. Общее тело $000B28.
BiosCopyToVDP    = $000ADE

; $FF29 — очистка спрайтов: 512 слов нулями по адресу $D000 (таблица
; атрибутов спрайтов), затем 40 слов VSRAM (вертикальная прокрутка).
BiosClearSprites = $0013D4

; $FF25 — вывод в таблицу имён: $0(a6) атрибут, который OR-ится в каждую
; запись, $2(a6) готовая команда VDP, $6(a6) источник. Рабочая база $FF0070.
BiosDrawTilemap  = $001608

; $FF2B — работа с прямоугольной областью: сохраняет и восстанавливает
; VDPAutoIncShadow, берёт из аргументов пару координат и сравнивает их.
; Что именно рисует — из одного тела не видно.
BiosBlitRegion   = $001A4C

; Теневой регистр 15 VDP (авто-инкремент) внутри VDPRegShadow:
; $FF0018 + 15*2. $FF2B сохраняет его перед сменой и возвращает обратно.
VDPAutoIncShadow = $FF0036

; ── Блок списка записей $FF9C7E..$FF9C84 ──────────────────────────────
; Три поля одной конструкции: курсор, счётчик и база.
; CurrentIndex — селектор, который читают вызываемые процедуры. Идиома в
; $007046: clr.b, вызвать четыре процедуры, addq.b #1, вызвать те же ещё
; раз, затем восстановить сохранённое значение. То есть один и тот же
; набор действий прогоняется для индекса 0, потом для 1. С литералами
; нигде не сравнивается, только с регистрами, поэтому верхнюю границу
; диапазона по коду не установить — «игрок» это или что-то иное, неясно.
CurrentIndex     = $FF9C82

; ── Индекс ресурсов ───────────────────────────────────────────────────
; Двухуровневый. BiosLoadAsset ($FF08) читает указатель из AssetRootPtr,
; тот ведёт на AssetIndexL1 — таблицу из 125 указателей на подтаблицы;
; в подтаблицах уже адреса самих блоков. Развёртка даёт 1629 блоков.
;
; ВАЖНО: у ВСЕХ 1629 блоков метод сжатия 1, а не 7. Раньше я оценил
; формат по девяти блокам, найденным через вызовы $FF04 в коде, — выборка
; оказалась непредставительной. Значит для распаковки графики нужен
; DecompressMethod1, и он один закрывает практически все ресурсы.
; Заголовок блока одинаков: 02 00 01 00 — размер $0200 (512 байт = 16
; тайлов 4bpp), затем байт метода.
AssetRootPtr     = $078810
AssetIndexL1     = $078818
BiosLoadAsset    = $001834
DecompressMethod1 = $000D68

; ── Ещё точки входа ───────────────────────────────────────────────────
; $FF02 — передача через DMA: ставит бит 4 (DMAEn) в теневом регистре 1
; и пишет его в VDP.
BiosDMATransfer  = $000C98
VDPMode2Shadow   = $FF001A   ; VDPRegShadow + 1*2, регистр 1 (Mode Register 2)

; $FF09 — интерпретатор байтового потока: читает (a2)+ и сравнивает с $F0,
; то есть значения от $F0 — команды. Похоже на скрипт или анимацию.
BiosRunScript    = $0018CA
